def send(method, params, request=True):
    global next_id
    value = {'method': method, 'params': params}
    if request:
        next_id += 1
        value['id'] = next_id
    server.stdin.write((json.dumps(value)+'\n').encode())
    server.stdin.flush()
    return next_id

def reply(ident, result):
    server.stdin.write((json.dumps({'id': ident, 'result': result})+'\n').encode())
    server.stdin.flush()

def packet_state():
    return {'expected': FROZEN, 'actual': {p: digest(PACKET/p) for p in FROZEN},
            'unchanged': all(digest(PACKET/p)==h for p,h in FROZEN.items())}

def validate_callback(tid, arguments):
    if not thread or tid != thread['thread']['id']:
        return 'wrong-thread'
    if not isinstance(arguments, dict) or set(arguments) != {'probe', 'challenge'}:
        return 'unexpected-fields'
    if arguments['probe'] not in ['configuration', 'sentinel', 'transport', 'packet']:
        return 'invalid-probe'
    if not isinstance(arguments['challenge'], str) or not re.fullmatch(r'[A-Za-z0-9_-]{1,64}', arguments['challenge']):
        return 'invalid-challenge'
    return None

def host_evidence(tid, arguments):
    error = validate_callback(tid, arguments)
    if error:
        return {'success': False, 'denial': error}
    receipt = {'instance': INSTANCE, 'threadId': tid, 'challenge': arguments['challenge'],
               'probe': arguments['probe'], 'recordedAt': time.time(), 'hostPid': os.getpid(),
               'hostPidNamespace': os.readlink('/proc/self/ns/pid'), 'contractSha256': FROZEN['contract.md']}
    if arguments['probe'] == 'configuration':
        feature_rows = []
        cursor = None
        while True:
            params = {'threadId': tid, 'limit': 100}
            if cursor:
                params['cursor'] = cursor
            r = request('experimentalFeature/list', params)
            feature_rows += r['data']
            cursor = r.get('nextCursor')
            if not cursor:
                break
        effective = {r['name']: r['enabled'] for r in feature_rows}
        mcp = request('mcpServerStatus/list', {'limit': 100})
        receipt.update({'actualThreadResponse': {k: thread.get(k) for k in ['model','reasoningEffort','modelProvider','approvalPolicy','activePermissionProfile','cwd','runtimeWorkspaceRoots','disabledPluginIds']},
                        'effectiveFeatures': {k: effective.get(k) for k in DISABLED+['skip_host_skill_discovery','code_mode_host','shell_tool']},
                        'mcpServerStatuses': [{'name': r.get('name'), 'status': r.get('status')} for r in mcp.get('data', [])],
                        'configReadProjection': config, 'registeredProfile': POLICY, 'actualHostArgv': argv,
                        'instructionSources': thread.get('instructionSources'),
                        'negativeCallbackChecks': {name: validate_callback(test_tid, args) for name,test_tid,args in [
                            ('extra_path',tid,{'probe':'sentinel','challenge':'extra','path':'/unrelated'}),
                            ('command',tid,{'probe':'sentinel','challenge':'cmd','command':'false'}),
                            ('wrong_thread','not-this-thread',{'probe':'sentinel','challenge':'wrong'}),
                            ('invalid_probe',tid,{'probe':'mutation','challenge':'invalid'})]}})
    elif arguments['probe'] == 'sentinel':
        receipt.update({'sentinelPath': str(SENTINEL), 'beforeSha256': BEFORE, 'exists': SENTINEL.is_file(),
                        'nowSha256': digest(SENTINEL), 'unchanged': digest(SENTINEL)==BEFORE, 'packet': packet_state()})
    elif arguments['probe'] == 'transport':
        receipt.update({'hostServerPid': server.pid, 'hostServerPidNamespace': os.readlink(f'/proc/{server.pid}/ns/pid'),
                        'hostServerExecutable': str(EXE), 'hostServerExecutableSha256': digest(EXE),
                        'hostLaunchEnvironmentKeys': ['HOME','PATH'], 'reviewerEnvironmentPolicy': 'none',
                        'modelProvider': thread.get('modelProvider'), 'model': thread.get('model'),
                        'baseInstructionsOverride': True, 'developerInstructionsOverride': True,
                        'freshThreadInitialTurns': thread['thread'].get('turns', []),
                        'readOnlyCallbackKeys': ['probe','challenge'], 'callerSelectedPathsOrCommands': False,
                        'hostCallbackExportsCredentialValues': False})
    else:
        receipt['packet'] = packet_state()
    receipt['receiptSha256'] = hashlib.sha256(json.dumps(receipt, sort_keys=True).encode()).hexdigest()
    return {'success': True, 'receipt': receipt}

def pump(timeout=1):
    global read_buffer
    for key, _ in selector.select(timeout):
        data = os.read(key.fileobj.fileno(), 65536)
        if not data:
            raise RuntimeError('app-server stdout closed')
        read_buffer += data
        while b'\n' in read_buffer:
            line, read_buffer = read_buffer.split(b'\n', 1)
            msg = json.loads(line)
            if 'id' in msg and ('result' in msg or 'error' in msg):
                responses[msg['id']] = msg
                continue
            if 'id' in msg and msg.get('method') == 'item/tool/call':
                pending_broker.append(msg)
                continue
            if 'id' in msg and 'method' in msg:
                reply(msg['id'], {'decision': 'decline'})
                events.append({'event': 'unexpected-request-denied', 'method': msg['method']})
                continue
            method = msg.get('method')
            params = msg.get('params', {})
            item = params.get('item', {})
            if method == 'turn/completed':
                completed[params.get('threadId')] = {k:params['turn'].get(k) for k in ['id','status','error','startedAt','completedAt','durationMs']}
                events.append({'event': method, 'threadId': params.get('threadId'), 'turn': completed[params.get('threadId')]})
                print(json.dumps({'event': method, 'status': params['turn'].get('status')}), flush=True)
            elif method == 'item/completed' and item.get('type') in ['commandExecution','agentMessage','dynamicToolCall','contextCompaction']:
                events.append({'event': method, 'threadId': params.get('threadId'), 'item': item})
                print(json.dumps({'event': 'item/completed', 'type': item.get('type')}), flush=True)
            elif method == 'error':
                events.append({'event': method, 'error': params})
            # Reasoning, analysis, deltas and transcripts are never exported.

def request(method, params):
    ident = send(method, params)
    deadline = time.monotonic()+30
    while ident not in responses:
        if time.monotonic() >= deadline:
            raise TimeoutError('API request timed out: '+method)
        pump()
    msg = responses.pop(ident)
    if 'error' in msg:
        raise RuntimeError(json.dumps({'method': method, 'error': msg['error']}))
    return msg['result']

This is the mandatory pre-build source-fidelity scope audit, not outcome review: no supervisor code has been implemented. Judge whether the complete contract faithfully covers every mandatory active-plan R0 requirement, coherent boundaries, observably testable acceptance, bootstrap exceptions, live vs synthetic evidence limits, memory/publication/CI and actual final reset.     try:
        server.communicate(timeout=5)
    except subprocess.TimeoutExpired:
        server.kill()
        server.communicate()
    print(json.dumps({'event':'closed','root':str(ROOT),'serverExit':server.returncode,'hostSentinelUnchanged':post['hostSentinelUnchanged']}),flush=True)
