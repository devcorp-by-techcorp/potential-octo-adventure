def send(method, params, request=True):
    global next_id
    value = {'method': method, 'params': params}
    if request:
        next_id += 1
        value['id'] = next_id
    server.stdin.write((json.dumps(value)+'\n').encode())
    server.stdin.flush()
    return next_id

def reply(ident, result):
    server.stdin.write((json.dumps({'id': ident, 'result': result})+'\n').encode())
    server.stdin.flush()

def packet_state():
    return {'expected': FROZEN, 'actual': {p: digest(PACKET/p) for p in FROZEN},
            'unchanged': all(digest(PACKET/p)==h for p,h in FROZEN.items())}

def validate_callback(tid, arguments):
    if not thread or tid != thread['thread']['id']:
        return 'wrong-thread'
    if not isinstance(arguments, dict) or set(arguments) != {'probe', 'challenge'}:
        return 'unexpected-fields'
    if arguments['probe'] not in ['configuration', 'sentinel', 'transport', 'packet']:
        return 'invalid-probe'
    if not isinstance(arguments['challenge'], str) or not re.fullmatch(r'[A-Za-z0-9_-]{1,64}', arguments['challenge']):
        return 'invalid-challenge'
    return None

def pump(timeout=1):
    global read_buffer
    for key, _ in selector.select(timeout):
        data = os.read(key.fileobj.fileno(), 65536)
        if not data:
            raise RuntimeError('app-server stdout closed')
        read_buffer += data
        while b'\n' in read_buffer:
            line, read_buffer = read_buffer.split(b'\n', 1)
            msg = json.loads(line)
            if 'id' in msg and ('result' in msg or 'error' in msg):
                responses[msg['id']] = msg
                continue
            if 'id' in msg and msg.get('method') == 'item/tool/call':
                pending_broker.append(msg)
                continue
            if 'id' in msg and 'method' in msg:
                reply(msg['id'], {'decision': 'decline'})
                events.append({'event': 'unexpected-request-denied', 'method': msg['method']})
                continue
            method = msg.get('method')
            params = msg.get('params', {})
            item = params.get('item', {})
            if method == 'turn/completed':
                completed[params.get('threadId')] = params['turn']
                events.append({'event': method, 'threadId': params.get('threadId'), 'turn': params['turn']})
                print(json.dumps({'event': method, 'status': params['turn'].get('status')}), flush=True)
            elif method == 'item/completed' and item.get('type') in ['commandExecution','agentMessage','dynamicToolCall','contextCompaction']:
                events.append({'event': method, 'threadId': params.get('threadId'), 'item': item})
                print(json.dumps({'event': 'item/completed', 'type': item.get('type')}), flush=True)
            elif method == 'error':
                events.append({'event': method, 'error': params})
            # Reasoning, analysis, deltas and transcripts are never exported.

def request(method, params):
    ident = send(method, params)
    deadline = time.monotonic()+30
    while ident not in responses:
        if time.monotonic() >= deadline:
            raise TimeoutError('API request timed out: '+method)
        pump()
    msg = responses.pop(ident)
    if 'error' in msg:
        raise RuntimeError(json.dumps({'method': method, 'error': msg['error']}))
    return msg['result']

def cli(args, cwd=None):
    p = subprocess.run(args, cwd=cwd, capture_output=True, text=True, timeout=40)
    if p.returncode:
        raise RuntimeError(json.dumps({'command':args[:3], 'exit':p.returncode, 'error':p.stderr[:300]}))
    return p.stdout.strip()

def wait_turn(tid, timeout=420):
    deadline=time.monotonic()+timeout
    while tid not in completed:
        if time.monotonic()>deadline: raise TimeoutError('bounded qualification turn did not complete')
        pump()
        while pending_broker:
            msg=pending_broker.pop(0);params=msg['params']
            value=host_evidence(params.get('threadId'),params.get('arguments')) if params.get('tool')=='bootstrap_evidence' else {'success':False,'denial':'unknown-tool'}
            events.append({'event':'bootstrap-evidence-response','threadId':params.get('threadId'),'input':params.get('arguments'),'response':value})
            reply(msg['id'],{'success':value.get('success',False),'contentItems':[{'type':'inputText','text':json.dumps(value)}]})
            print(json.dumps({'event':'bootstrap-evidence','probe':params.get('arguments',{}).get('probe'),'success':value.get('success')}),flush=True)
    if completed[tid].get('status')!='completed': raise RuntimeError('qualification turn failed')

def start_fresh(label, checkpoint):
    global thread
    tool={'type':'function','name':'bootstrap_evidence','description':'Fixed read-only actual Empirical, Git/CI and source/compaction evidence. No caller-selected paths, commands, queries, writes or credentials.',
          'inputSchema':{'type':'object','additionalProperties':False,'properties':{'probe':{'type':'string','enum':['empirical','git_ci','source_lifecycle']},'challenge':{'type':'string','pattern':'^[A-Za-z0-9_-]{1,64}$'}},'required':['probe','challenge']}}
    thread=request('thread/start',{'model':'gpt-6.1-sol','allowProviderModelFallback':False,'permissions':'r0_bootstrap_memory','approvalPolicy':'never','cwd':str(OUTPUT),'ephemeral':True,'dynamicTools':[tool],
                   'baseInstructions':'You are a fresh Studio Suite bootstrap controller. Your first turn is memory-only: call bootstrap_evidence with probe empirical and an ASCII letters/digits/underscore-only challenge such as r0_bootstrap. On denial retry memory only; do not read any file. Return MEMORY_READY only after successful live memory/workspace/policy retrieval. Filesystem source reads are not granted until the host validates that turn. In the next turn read the checkpoint and indicated qualification contract. Independently verify current authority hashes and source/Git/CI facts. Do not export reasoning or credential values. Missing evidence or disagreement blocks advancement. No application changes or remote writes.',
                   'developerInstructions':'This is read-only R0 preparation. R0 is not GREEN. Do not claim RESET_CONFIRMED or start any product phase. Return factual JSON with every applicable BQ row, prior gate state, exact next scope, advancement permission and limits. Prior controller messages, compacted items and previous verdicts are unavailable.'})
    tid=thread['thread']['id']
    if tid==predecessor_id or thread['thread'].get('turns') or thread.get('model')!='gpt-6.1-sol' or thread.get('reasoningEffort')!='medium':
        raise RuntimeError('fresh controller identity/model/history mismatch')
    events.append({'event':'fresh-thread-start','label':label,'response':{k:thread.get(k) for k in ['model','reasoningEffort','activePermissionProfile','instructionSources']},'threadId':tid,'initialTurns':thread['thread'].get('turns',[])})
    prompt=f'Checkpoint pointer: {checkpoint}\nNext contract ID: R0 preparatory bootstrap qualification.'
    events.append({'event':'fresh-input','threadId':tid,'text':prompt})
    request('turn/start',{'threadId':tid,'model':'gpt-6.1-sol','effort':'medium','input':[{'type':'text','text':prompt,'text_elements':[]}]})
    wait_turn(tid)
    successes=[e for e in events if e.get('event')=='bootstrap-evidence-response' and e.get('threadId')==tid and e.get('input',{}).get('probe')=='empirical' and e['response'].get('success')]
